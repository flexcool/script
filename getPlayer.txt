function getPlayerHTML(object) {
        let videoList = object.related || [];
        const title = (object.name || '').replace(/'/g, "\\'").replace(/"/g, '&quot;');
        const remark = (object.remark || '').replace(/'/g, "\\'").replace(/"/g, '&quot;');
        const totalCount = videoList.length;
        
        // 按每50集分组
        const GROUP_SIZE = 50;
        const groups = [];
        for (let i = 0; i < videoList.length; i += GROUP_SIZE) {
            groups.push(videoList.slice(i, i + GROUP_SIZE));
        }
        
        // 生成分组HTML
        let groupsHtml = '';
        for (let g = 0; g < groups.length; g++) {
            const group = groups[g];
            const startNum = g * GROUP_SIZE + 1;
            const endNum = Math.min((g + 1) * GROUP_SIZE, totalCount);
            
            // 生成该组的剧集按钮
            let groupEpisodesHtml = '';
            for (let i = 0; i < group.length; i++) {
                const episode = group[i];
                const globalIndex = g * GROUP_SIZE + i;
                let displayName = episode.name.replace(/[第集话]/g, '') || String(globalIndex + 1);
                let safeUrl = episode.url.replace(/'/g, "\\'").replace(/"/g, '&quot;');
                groupEpisodesHtml += `<div class="episode-item" data-url="${safeUrl}" data-index="${globalIndex}">${displayName}</div>`;
            }
            
            // 默认第一组展开，其他折叠
            const isExpanded = (g === 0) ? 'expanded' : '';
            groupsHtml += `
                <div class="episode-group ${isExpanded}" data-group="${g}">
                    <div class="group-header" onclick="toggleGroup(${g})">
                        <span class="group-arrow">▶</span>
                        <span class="group-title">${startNum}-${endNum}集</span>
                        <span class="group-count">${group.length}</span>
                    </div>
                    <div class="group-content" id="group_${g}">
                        <div class="episode-grid">${groupEpisodesHtml}</div>
                    </div>
                </div>
            `;
        }

// HLS 播放器实例

// 隐藏错误提示
function hideError() {
    errorOverlay.style.display = 'none';

// 显示错误提示
function showError(msg) {
    errorOverlay.innerHTML = msg;
    errorOverlay.style.display = 'block';
    setTimeout(function() {
        if (errorOverlay.innerHTML === msg) {
            errorOverlay.style.display = 'none';

// 设置加载状态

// 清理旧的 HLS 实例
function destroyHls() {
    if (hlsInstance) {
        try {
            hlsInstance.destroy();
        } catch(e) {}
        hlsInstance = null;
    }
}

// 使用 HLS.js 播放 m3u8
function playM3u8WithHls(url) {
    return new Promise(function(resolve, reject) {
        // 清理旧实例
        destroyHls();
        
        // 暂停当前播放
        videoPlayer.pause();
        videoPlayer.src = '';
        
        // 检查浏览器原生支持 (Safari)
        if (videoPlayer.canPlayType('application/vnd.apple.mpegurl')) {
            videoPlayer.src = url;
            videoPlayer.addEventListener('loadedmetadata', function onLoad() {
                videoPlayer.removeEventListener('loadedmetadata', onLoad);
                resolve();
            });
            videoPlayer.addEventListener('error', function onErr() {
                videoPlayer.removeEventListener('error', onErr);
                reject(new Error('原生播放失败'));
            });
            return;
        }
        
        // 使用 HLS.js
        if (Hls && Hls.isSupported()) {
            var config = {
                enableWorker: true,
                lowLatencyMode: false,
                maxBufferLength: 30,
                maxMaxBufferLength: 60,
                liveSyncDurationCount: 3,
                // 针对 Webview 环境优化
                xhrSetup: function(xhr, xhrUrl) {
                    // 如果 m3u8 地址跨域，可以添加 headers
                    // xhr.setRequestHeader('Origin', location.origin);
                }
            };
            
            hlsInstance = new Hls(config);
            hlsInstance.loadSource(url);
            hlsInstance.attachMedia(videoPlayer);
            
            // 监听就绪
            hlsInstance.on(Hls.Events.MANIFEST_PARSED, function() {
                resolve();
            });
            
            // 监听错误
            hlsInstance.on(Hls.Events.ERROR, function(event, data) {
                if (data.fatal) {
                    switch(data.type) {
                        case Hls.ErrorTypes.NETWORK_ERROR:
                            reject(new Error('网络错误: ' + (data.details || '')));
                            break;
                        case Hls.ErrorTypes.MEDIA_ERROR:
                            // 尝试恢复
                            if (hlsInstance) {
                                hlsInstance.recoverMediaError();
                                reject(new Error('媒体解码错误'));
                            }
                            break;
                        default:
                            reject(new Error('播放错误: ' + (data.details || '')));
                            break;
                    }
                } else {
                    // 非致命错误，仅记录
                    console.warn('HLS警告:', data.details);
                }
            });
            
        } else {
            reject(new Error('当前浏览器不支持 HLS 播放'));
        }
    });
}

// 判断是否为 m3u8 链接
function isM3u8Url(url) {
    if (!url) return false;
    var lowerUrl = url.toLowerCase();
    return lowerUrl.includes('.m3u8') || lowerUrl.includes('m3u8');
}

// 核心播放函数
async function playVideoByUrl(url, index) {
    hideError();
    setLoading(true);
    
    try {
        // 如果是 m3u8 链接，使用 HLS.js 播放
        if (isM3u8Url(url)) {
            await playM3u8WithHls(url);
            console.log('HLS.js 解析成功:', url);
        } else {
            // 非 m3u8 链接（如 mp4），降级为直接播放
            destroyHls();
            videoPlayer.src = url;
            await new Promise(function(resolve, reject) {
                videoPlayer.addEventListener('canplay', function onCanPlay() {
                    videoPlayer.removeEventListener('canplay', onCanPlay);
                    resolve();
                });
                videoPlayer.addEventListener('error', function onErr() {
                    videoPlayer.removeEventListener('error', onErr);
                    reject(new Error('无法加载视频'));
                });
                videoPlayer.load();
            });
        }
        
        // 更新激活样式
        var allItems = document.querySelectorAll('.episode-item');
        for (var i = 0; i < allItems.length; i++) {
            allItems[i].classList.remove('active');
        }
        var targetItem = document.querySelector('.episode-item[data-index="' + index + '"]');
        if (targetItem) targetItem.classList.add('active');
        
        // 尝试自动播放
        try {
            await videoPlayer.play();
        } catch(playErr) {
            console.warn('自动播放被阻止，需要用户交互:', playErr);
        }
        
    } catch(err) {
        console.error('播放失败:', err);
        showError('播放失败: ' + (err.message || '未知错误'));
        
        // 可选：降级到解析接口
        // fallbackToParser(url, index);
    } finally {
        setLoading(false);
    }
}

// 可选：降级方案 - 使用第三方解析接口
function fallbackToParser(url, index) {
    var parserUrl = '';
    if (url.includes('lz-cdn') || url.includes('cdnlz')) {
        parserUrl = 'https://lziplayer.com/?url=' + encodeURIComponent(url);
    } else if (isM3u8Url(url)) {
        parserUrl = 'https://www.iqiyizyjx.com/?url=' + encodeURIComponent(url);
    }
    if (parserUrl) {
        console.log('降级使用解析接口:', parserUrl);
        destroyHls();
        videoPlayer.src = parserUrl;
        videoPlayer.play().catch(function(e) {});
    }
}

    var idx = parseInt(episodeItems[i].getAttribute('data-index'));
    allUrls[idx] = url;
    episodeItems[i].addEventListener('click', (function(url, idx) {
        return function() { 
            currentIndex = idx;
            playVideoByUrl(url, idx); 
        };
    })(url, idx));
}

function playNext() {
    var nextIndex = (currentIndex + 1);
    if (nextIndex < allUrls.length && allUrls[nextIndex]) {
        currentIndex = nextIndex;
        playVideoByUrl(allUrls[currentIndex], currentIndex);
    }
}

videoPlayer.onended = function() {
    playNext();
};
